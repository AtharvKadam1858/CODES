# compare their outputs for data integrity verification.
# ============================================================

import hashlib


# ============================================================
# Function to Generate Hash
# ============================================================

def generate_hash(data, algorithm):

    if algorithm == "MD5":
        return hashlib.md5(data.encode()).hexdigest()

    elif algorithm == "SHA1":
        return hashlib.sha1(data.encode()).hexdigest()

    elif algorithm == "SHA224":
        return hashlib.sha224(data.encode()).hexdigest()

    elif algorithm == "SHA256":
        return hashlib.sha256(data.encode()).hexdigest()

    elif algorithm == "SHA384":
        return hashlib.sha384(data.encode()).hexdigest()

    elif algorithm == "SHA512":
        return hashlib.sha512(data.encode()).hexdigest()


# ============================================================
# Algorithms List
# ============================================================

algorithms = {
    "MD5": 32,
    "SHA1": 40,
    "SHA224": 56,
    "SHA256": 64,
    "SHA384": 96,
    "SHA512": 128
}


# ============================================================
# Original Data
# ============================================================

original_data = "DataIntegrity123"

print("=" * 80)
print("HASHING ALGORITHMS FOR DATA INTEGRITY")
print("=" * 80)

print("\nOriginal Data :", original_data)

print("\nGenerated Hashes")
print("-" * 80)

for algo, length in algorithms.items():

    h = generate_hash(original_data, algo)

    print(f"\n{algo} ({length} Characters)")
    print(h)


# ============================================================
# Modified Data
# ============================================================

modified_data = "DataIntegrity124"

print("\n" + "=" * 80)
print("AFTER MODIFYING THE DATA")
print("=" * 80)

print("\nModified Data :", modified_data)

print("\nGenerated Hashes")
print("-" * 80)

for algo, length in algorithms.items():

    h = generate_hash(modified_data, algo)

    print(f"\n{algo} ({length} Characters)")
    print(h)


# ============================================================
# Hash Comparison
# ============================================================

print("\n" + "=" * 80)
print("HASH COMPARISON")
print("=" * 80)

for algo in algorithms:

    original_hash = generate_hash(original_data, algo)
    modified_hash = generate_hash(modified_data, algo)

    print(f"\n{algo}")

    if original_hash == modified_hash:
        print("Status : SAME HASH")
    else:
        print("Status : DIFFERENT HASH")
        print("Reason : One character change completely changes the hash.")
        print("Avalanche Effect Demonstrated.")
