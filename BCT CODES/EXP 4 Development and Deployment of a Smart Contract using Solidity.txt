contract SimpleStorage {

    uint256 private storedData;

    // Store a value
    function set(uint256 _value) public {
        storedData = _value;
    }

    // Retrieve stored value
    function get() public view returns (uint256) {
        return storedData;
    }

    // Increase stored value
    function increment() public {
        storedData++;
    }

    // Decrease stored value
    function decrement() public {
        require(storedData > 0, "Value cannot be negative");
        storedData--;
    }
}

import hashlib
import datetime


# -----------------------------
# Smart Contract Simulation
# -----------------------------
class SimpleStorage:

    def __init__(self):

        self.stored_data = 0
        self.owner = "Admin"
        self.deployed = True

        print("Smart Contract Deployed Successfully!")


    # Store a value
    def set(self, value, sender):

        if sender != self.owner:
            print("Access Denied! Only owner can store data.")
            return

        self.stored_data = value
        print("Value Stored Successfully:", value)


    # Retrieve a value
    def get(self):

        print("Stored Value:", self.stored_data)
        return self.stored_data


    # Increment value
    def increment(self, sender):

        if sender != self.owner:
            print("Access Denied!")
            return

        self.stored_data += 1
        print("Value After Increment:", self.stored_data)


    # Decrement value
    def decrement(self, sender):

        if sender != self.owner:
            print("Access Denied!")
            return

        if self.stored_data > 0:
            self.stored_data -= 1
            print("Value After Decrement:", self.stored_data)
        else:
            print("Value Cannot Be Negative!")
