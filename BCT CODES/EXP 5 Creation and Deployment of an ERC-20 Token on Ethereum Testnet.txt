class ERC20Token:

    def __init__(self, name, symbol, supply, owner):

        self.name = name
        self.symbol = symbol
        self.total_supply = supply

        self.balances = {
            owner: supply
        }

        self.allowances = {}

        print("\nToken Created Successfully")
        print("Name :", self.name)
        print("Symbol :", self.symbol)
        print("Total Supply :", self.total_supply)


    def balanceOf(self, address):

        return self.balances.get(address, 0)


    def transfer(self, sender, receiver, amount):

        if self.balanceOf(sender) >= amount:

            self.balances[sender] -= amount

            self.balances[receiver] = (
                self.balanceOf(receiver) + amount
            )

            print("\nTransfer Successful")

        else:

            print("\nInsufficient Balance")


    def approve(self, owner, spender, amount):

        self.allowances[(owner, spender)] = amount

        print("\nApproval Successful")


    def allowance(self, owner, spender):

        return self.allowances.get(
            (owner, spender),
            0
        )


    def transferFrom(self, spender, owner, receiver, amount):

        allowed = self.allowance(owner, spender)

        if allowed >= amount and self.balanceOf(owner) >= amount:

            self.balances[owner] -= amount

            self.balances[receiver] = (
                self.balanceOf(receiver) + amount
            )

            self.allowances[(owner, spender)] -= amount

            print("\nTransferFrom Successful")

        else:

            print("\nTransfer Failed")



# ============================================================

owner = "Alice"
