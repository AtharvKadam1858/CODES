# -----------------------------
# Generate N Even Numbers
# -----------------------------
def generate_even_numbers(n):

    print("\n--- FIRST", n, "EVEN NUMBERS ---")

    even_numbers = []

    for i in range(1, n + 1):

        even_numbers.append(2 * i)

    print("Even Numbers:", even_numbers)


# -----------------------------
# Matrix Multiplication
# -----------------------------
def multiply_matrices(A, B):

    rows_A = len(A)
    cols_A = len(A[0])

    rows_B = len(B)
    cols_B = len(B[0])

    if cols_A != rows_B:

        print("Matrix Multiplication Not Possible!")
        return None

    result = [
        [0 for j in range(cols_B)]
        for i in range(rows_A)
    ]

    for i in range(rows_A):

        for j in range(cols_B):

            for k in range(cols_A):

                result[i][j] += A[i][k] * B[k][j]

    return result


# -----------------------------
# Main Program
# -----------------------------
print("=" * 60)
print(" EVEN NUMBERS AND MATRIX MULTIPLICATION ")
print("=" * 60)

# Generate Even Numbers
n = 10

generate_even_numbers(n)


# Matrix A
A = [
    [1, 2],
    [3, 4]
]

# Matrix B
B = [
    [5, 6],
    [7, 8]
]

print("\n--- MATRIX A ---")

for row in A:
    print(row)

print("\n--- MATRIX B ---")

for row in B:
    print(row)


# Multiply Matrices
result = multiply_matrices(A, B)

print("\n--- MATRIX MULTIPLICATION RESULT ---")

if result is not None:

    for row in result:
        print(row)

print("\nProgram Executed Successfully.")
